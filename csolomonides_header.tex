% Defines a command that allows displaying the CV in multiple languages
\usepackage[T1]{fontenc} 
\usepackage[english, french]{babel}
\usepackage{ifthen}
\usepackage{hyperref}
\usepackage{enumitem}

\providecommand{\multilingual}[2]{
    \ifx\french\undefined
        #1
    \else
        \if #2\empty
            \textbf{INFORMATION MISSING. English text: ``#1'' \\}
        \else
            #2
        \fi
    \fi
}

\providecommand{\cdd}[1]{
    \ifx\cddrequested\undefined
        % NOP
    \else
        #1
    \fi
}
