% Defines a command that allows displaying the CV in multiple languages
\usepackage[english, french]{babel}
\usepackage[autolanguage]{numprint} % for the \nombre command
\usepackage{ifthen}
\usepackage{helvet}
\renewcommand{\familydefault}{\sfdefault}

% Define defaults instead of checking for absence
\newcommand{\cddrequested}{false}
\newcommand{\targetrole}{softwarengineer}
\newcommand{\cvlanguage}{english}

\providecommand{\multilingual}[2]{
    \ifthenelse{
        \equal{\cvlanguage}{english}
    }{
        % If true
        #1
    }{
        % if false
        \if #2\empty
            \textbf{INFORMATION MISSING. English text: ``#1''\\}
        \else
            #2
        \fi
    }
}

\providecommand{\cdd}[1]{
    \ifthenelse{\equal{\cddrequested}{true}}{#1}{}
}

\newenvironment{experiencebullets}{
    \begin{itemize}[label=-, noitemsep, topsep=0pt, partopsep=0pt, parsep=0pt]
}{
    \end{itemize}
}
